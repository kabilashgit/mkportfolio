🎯 How I Built It (Step-by-Step)

Step 1: Get Your Jira API Token

First, you need access to Jira's API. Super easy:

  1. Go to: id.atlassian.com/manage-profile/security/api-tokens
  2. Click "Create API token"
  3. Copy it and save it somewhere safe

Step 2: Set Up Environment Variables

Store your credentials securely in environment variables (never hardcode them!):

For Mac/Linux (add to ~/.bashrc):

export JIRA_INSTANCE_URL="https://yourcompany.atlassian.net"
export JIRA_AUTH_TOKEN="Basic $(echo -n 'your.email@company.com:YOUR_API_TOKEN' | base64)"

For Windows PowerShell (add to $PROFILE):

$env:JIRA_INSTANCE_URL = "https://yourcompany.atlassian.net"
$email = "your.email@company.com"
$apiToken = "YOUR_API_TOKEN"
$base64 = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes("${email}:${apiToken}"))
$env:JIRA_AUTH_TOKEN = "Basic $base64"

Step 3: Write the Fetcher Script

I created a Python script that does the heavy lifting. It:

  • ✓ Connects to Jira API
  • ✓ Downloads ticket data
  • ✓ Grabs all attachments automatically
  • ✓ Saves everything locally for caching
#!/usr/bin/env python3
"""
Fetch Jira ticket from API and download all attachments.
Usage: python fetch_ticket.py TICKET-KEY
"""

import sys
import os
import json
import urllib.request
import urllib.error

def fetch_ticket_json(ticket_key, instance_url, auth_token):
    """Fetch ticket JSON from Jira API."""
    url = f"{instance_url}/rest/api/3/issue/{ticket_key}"

    req = urllib.request.Request(url)
    req.add_header('Authorization', auth_token)
    req.add_header('Accept', 'application/json')

    try:
        with urllib.request.urlopen(req) as response:
            return json.loads(response.read().decode('utf-8'))
    except urllib.error.HTTPError as e:
        print(f"Error fetching ticket: HTTP {e.code}")
        sys.exit(1)

def download_attachment(url, output_path, auth_token):
    """Download a single attachment."""
    req = urllib.request.Request(url)
    req.add_header('Authorization', auth_token)

    try:
        with urllib.request.urlopen(req) as response:
            with open(output_path, 'wb') as f:
                f.write(response.read())
        return os.path.getsize(output_path)
    except Exception as e:
        print(f"  ✗ Failed: {e}")
        return 0

def main():
    if len(sys.argv) < 2:
        print("Usage: python fetch_ticket.py TICKET-KEY")
        sys.exit(1)

    ticket_key = sys.argv[1]

    # Get environment variables
    instance_url = os.environ.get('JIRA_INSTANCE_URL')
    auth_token = os.environ.get('JIRA_AUTH_TOKEN')

    if not instance_url or not auth_token:
        print("Error: JIRA_INSTANCE_URL and JIRA_AUTH_TOKEN must be set")
        sys.exit(1)

    # Determine save location (project cache)
    project_cache = os.path.join(os.getcwd(), '.claude/temp/jira', ticket_key)
    ticket_file = os.path.join(project_cache, 'ticket.json')
    attachments_dir = os.path.join(project_cache, 'attachments')

    # Create directories
    os.makedirs(attachments_dir, exist_ok=True)

    print(f"Fetching ticket {ticket_key} from Jira...")

    # Fetch ticket JSON
    ticket_data = fetch_ticket_json(ticket_key, instance_url, auth_token)

    # Save ticket JSON
    with open(ticket_file, 'w', encoding='utf-8') as f:
        json.dump(ticket_data, f, indent=2, ensure_ascii=False)

    print(f"✓ Ticket data saved to {ticket_file}")

    # Download attachments
    attachments = ticket_data.get('fields', {}).get('attachment', [])

    if attachments:
        print(f"\nDownloading {len(attachments)} attachment(s)...")

        for att in attachments:
            filename = att['filename']
            url = att['content']
            output_path = os.path.join(attachments_dir, filename)

            size = download_attachment(url, output_path, auth_token)
            if size > 0:
                size_str = f"{size / 1024:.1f} KB" if size < 1024 * 1024 else f"{size / (1024 * 1024):.1f} MB"
                print(f"  ✓ {filename} ({size_str})")
    else:
        print("\nNo attachments found")

    print(f"\n✓ Saved to: {project_cache}/")

if __name__ == '__main__':
    main()
#!/usr/bin/env python3
"""
Display cached Jira ticket information.
Usage: python display_ticket.py TICKET-KEY
"""

import sys
import os
import json
import time

def find_ticket(ticket_key):
    """Check both cache locations and return path if found."""
    project_cache = os.path.join(os.getcwd(), '.claude/temp/jira', ticket_key, 'ticket.json')
    user_cache = os.path.join(os.path.expanduser('~'), '.claude/temp/jira', ticket_key, 'ticket.json')

    if os.path.exists(project_cache):
        return project_cache, 'project'
    elif os.path.exists(user_cache):
        return user_cache, 'user'
    else:
        return None, None

def format_cache_info(ticket_path, cache_type, ticket_key):
    """Display cache information."""
    print(f'📁 Ticket {ticket_key} found in {cache_type} cache')

    age_seconds = time.time() - os.path.getmtime(ticket_path)
    age_minutes = age_seconds / 60

    location = 'project (.claude/temp/jira/)' if cache_type == 'project' else 'user profile (~/.claude/temp/jira/)'
    print(f'Location: {location}')

    if age_minutes < 60:
        print(f'Last fetched: {age_minutes:.0f} minutes ago')
    else:
        age_hours = age_minutes / 60
        print(f'Last fetched: {age_hours:.1f} hours ago')

    # Count attachments
    attachments_dir = os.path.dirname(ticket_path) + '/attachments'
    if os.path.exists(attachments_dir):
        attachments = [f for f in os.listdir(attachments_dir) if os.path.isfile(os.path.join(attachments_dir, f))]
        print(f'Cached attachments: {len(attachments)} file(s)')
    print()

def display_ticket(ticket_path):
    """Parse and display ticket details."""
    with open(ticket_path, encoding='utf-8') as f:
        data = json.load(f)

        print(f"**Key:** {data['key']}")
        print(f"**Summary:** {data['fields']['summary']}")
        print(f"**Status:** {data['fields']['status']['name']}")
        print(f"**Type:** {data['fields']['issuetype']['name']}")
        print(f"**Priority:** {data['fields']['priority']['name']}")

        assignee = data['fields'].get('assignee')
        print(f"**Assignee:** {assignee['displayName'] if assignee else 'Unassigned'}")

        # Show attachments
        attachments = data['fields'].get('attachment', [])
        if attachments:
            print(f"\n**Attachments ({len(attachments)} files):**")
            for att in attachments:
                print(f"  ✓ {att['filename']} - Cached")

def main():
    if len(sys.argv) < 2:
        print("Usage: python display_ticket.py TICKET-KEY")
        sys.exit(1)

    ticket_key = sys.argv[1]

    # Find ticket
    ticket_path, cache_type = find_ticket(ticket_key)

    if not ticket_path:
        print(f"Ticket {ticket_key} not found in cache")
        sys.exit(0)

    # Display cache info
    format_cache_info(ticket_path, cache_type, ticket_key)

    # Display ticket details
    display_ticket(ticket_path)

if __name__ == '__main__':
    main()

# Jira Ticket Fetcher

Fetches Jira tickets with all attachments and caches them for reuse.

## Usage

`/jira-fetch TICKET-123` or just mention "fetch jira ticket PROJECT-456"

## What it does

1. Checks local cache first (project and user-level)
2. If not cached, fetches from Jira API
3. Downloads all ticket attachments
4. Saves everything locally
5. Displays formatted ticket information

## Requirements

Environment variables must be set:
- `JIRA_INSTANCE_URL` - Your Jira URL
- `JIRA_AUTH_TOKEN` - Base64 encoded credentials

## Files

- `fetch_ticket.py` - Main fetcher script
- `display_ticket.py` - Formatting helper

## Cache Locations

- `.claude/temp/jira/` - Project-level cache
- `~/.claude/temp/jira/` - User-level cache
